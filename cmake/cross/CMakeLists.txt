# Cross-compile check: compiles libobs for another OS from this host, with the
# LLVM and SDK soldr provisions (cmake/common/soldr.cmake) and the same
# prebuilt dependencies that OS's real build uses (CMakePresets.json, .deps/).
# It compiles every libobs source that OS builds, both the C build's ported
# files and the Rust build's C halves, and links nothing.
#
#   cmake -S cmake/cross -B build_cross_win-x64 -G Ninja -DOBS_CROSS_TARGET=win-x64
#   cmake --build build_cross_win-x64
#
# or, from a configured OBS build: cmake --build <build> --target cross-win-x64
# (cross-all for every alias). Aliases: win-x64, mac-arm64, mac-x64.
# Warnings are reported; -DOBS_CROSS_WERROR=ON makes them errors. PR CI runs
# it (build-project.yaml, Cross Compile) when a change touches Windows or
# macOS host code in a component listed here (ci/ci_scope.py, #1294).

cmake_minimum_required(VERSION 3.28...3.30)

# The Linux-side xwin SDK ships only the release CRT (there is no msvcrtd.lib),

# Debug. Set before project() so the configuration flags are computed for it.

  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

function(_obs_cross_object_library name)
  set(files ${ARGN})
  list(FILTER files INCLUDE REGEX "\\.(c|m|cpp)$")
  list(TRANSFORM files PREPEND "${OBS_SOURCE_ROOT}/libobs/")
  add_library(${name} OBJECT ${files})
  target_include_directories(
    ${name}
    PRIVATE
      "${OBS_SOURCE_ROOT}/libobs"
      "${CMAKE_BINARY_DIR}/config"
      "${OBS_SOURCE_ROOT}/deps/libcaption"
      "${OBS_CROSS_DEPS}/include"
      $<$<STREQUAL:${_host},WINDOWS>:${OBS_SOURCE_ROOT}/deps/w32-pthreads>
  )
  target_compile_definitions(
    ${name}
    PRIVATE
      IS_LIBOBS
      ENABLE_HEVC
      $<$<STREQUAL:${_host},WINDOWS>:NOMINMAX;UNICODE;_UNICODE;_CRT_SECURE_NO_WARNINGS;_CRT_NONSTDC_NO_WARNINGS>
  )
  target_compile_options(${name} PRIVATE "$<$<COMPILE_LANGUAGE:C,OBJC>:${_obs_clang_c_options}>")
  set_target_properties(
    ${name}
    PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED TRUE COMPILE_WARNING_AS_ERROR ${OBS_CROSS_WERROR}
  )
endfunction()

# Opt-in link step (OBS_CROSS_LINK): a real shared libobs for the alias from
# the same sources the shipped Rust build links, against the prebuilt
# dependencies' import libraries. This turns "compiles for Windows/macOS from
# Linux" into "links for Windows/macOS from Linux", so a link error stops
# living only on the native runners (#3059). The default stays compile-only.
#
# The link object set is the Rust build's, not the C reference's: libobs's
# C ported lists are compiled here for the C-side check but production links
# Rust for them (libobs/cmake/sources.cmake), so linking both would collide
# (base.c and base-variadic.c both define blogva).
function(_obs_cross_shared_libobs name)
  set(_sources ${LIBOBS_SOURCES} ${LIBOBS_${_host}_SOURCES} ${LIBOBS_RUST_SOURCES})
  _obs_cross_object_library(${name}-objects ${_sources})
  add_library(${name} SHARED)
  target_link_libraries(${name} PRIVATE ${name}-objects)
  # The production hybrid links the Rust halves from libobs-rust, so the cross
  # link needs that staticlib too: without it the ported C halves' shims stay
  # undefined. Build it with soldr cargo for the alias's triple.
  if(OBS_CROSS_TARGET STREQUAL "win-x64")
    set(_rust_triple "x86_64-pc-windows-msvc")
    set(_rust_archive "${OBS_SOURCE_ROOT}/target/${_rust_triple}/release/libobs_rust.lib")
  elseif(OBS_CROSS_TARGET STREQUAL "mac-arm64")
    set(_rust_triple "aarch64-apple-darwin")
    set(_rust_archive "${OBS_SOURCE_ROOT}/target/${_rust_triple}/release/liblibobs_rust.a")
  elseif(OBS_CROSS_TARGET STREQUAL "mac-x64")
    set(_rust_triple "x86_64-apple-darwin")
    set(_rust_archive "${OBS_SOURCE_ROOT}/target/${_rust_triple}/release/liblibobs_rust.a")
  endif()
  add_custom_command(
    OUTPUT "${_rust_archive}"
    COMMAND soldr cargo build --release --target "${_rust_triple}" -p libobs-rust
    WORKING_DIRECTORY "${OBS_SOURCE_ROOT}"
    COMMENT "Building libobs-rust staticlib for ${_rust_triple}"
    DEPENDS "${OBS_SOURCE_ROOT}/Cargo.toml" "${OBS_SOURCE_ROOT}/Cargo.lock"
    USES_TERMINAL
  )
  add_custom_target(${name}-rust DEPENDS "${_rust_archive}")
  add_dependencies(${name} ${name}-rust)
  # Whole archive: libobs itself does not call every shim (plugins do), and the
  # DLL's export set must keep them.
  target_link_libraries(${name} PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,${_rust_archive}>")
  target_link_directories(${name} PRIVATE "${OBS_CROSS_DEPS}/lib")
  set_target_properties(
    ${name}
    PROPERTIES
      C_STANDARD 17
      C_STANDARD_REQUIRED TRUE
      CXX_STANDARD 17
      CXX_STANDARD_REQUIRED TRUE
      PREFIX ""
      OUTPUT_NAME obs
  )
  # The libraries libobs links on its own platform
  # (libobs/CMakeLists.txt:121-138, libobs/cmake/os-*.cmake). FFmpeg and
  # jansson come from the prebuilt dependencies; the OS import libraries are
  # lower case in the Linux-side xwin SDK (a Windows host finds them either way).
  set(
    _common_deps
    avcodec
    avformat
    avutil
    swscale
    swresample
    jansson
  )
  if(_host STREQUAL WINDOWS)
    # The Rust standard library pulls winsock and NT APIs in through its
    # objects; the shipped libobs link gets these from the same system libs.
    target_link_libraries(
      ${name}
      PRIVATE ${_common_deps} avrt dwmapi dxgi winmm rpcrt4 ws2_32 ntdll bcrypt userenv
    )
    # The Rust staticlib's object directives request MSVCRT.lib / OLDNAMES.lib
    # with Windows' case, while the Linux-side xwin SDK ships msvcrt.lib /
    # oldnames.lib in lower case. Link against upper-case aliases so the
    # directives resolve on a case-sensitive host.
    if(OBS_CROSS_TARGET STREQUAL "win-x64")
      set(_crt_arch x86_64)
    else()
      set(_crt_arch aarch64)
    endif()
    set(_crt_lib "${CMAKE_FIND_ROOT_PATH}/crt/lib/${_crt_arch}")
    set(_case_lib_dir "${CMAKE_BINARY_DIR}/cross-case-libs")
    file(MAKE_DIRECTORY "${_case_lib_dir}")
    foreach(_lib MSVCRT.lib OLDNAMES.lib)
      string(TOLOWER "${_lib}" _lower)
      execute_process(
        COMMAND "${CMAKE_COMMAND}" -E create_symlink "${_crt_lib}/${_lower}" "${_case_lib_dir}/${_lib}"
      )
    endforeach()
    target_link_directories(${name} PRIVATE "${_case_lib_dir}")
    # Rust no_mangle shims in a staticlib are not dllexport, so name them, as
    # libobs/cmake/rust-os-windows.cmake does for the native build.
    file(GLOB _obs_rust_export_fragments "${OBS_SOURCE_ROOT}/libobs/cmake/rust-exports.d/*.txt")
    foreach(_fragment IN LISTS _obs_rust_export_fragments)
      file(STRINGS "${_fragment}" _obs_rust_exports REGEX "^[A-Za-z_]")
      foreach(_sym IN LISTS _obs_rust_exports)
        if(_sym MATCHES " unix$")
          continue()
        endif()
        target_link_options(${name} PRIVATE "LINKER:/EXPORT:${_sym}")
      endforeach()
    endforeach()
    # w32-pthreads is its own DLL in the shipped build
    # (deps/w32-pthreads/CMakeLists.txt: add_library(w32-pthreads SHARED)), so
    # its source and its DllMain stay in a separate shared library here too.
    add_library(${name}-pthreads SHARED "${OBS_SOURCE_ROOT}/deps/w32-pthreads/pthread.c")
    target_include_directories(
      ${name}-pthreads
      PRIVATE "${OBS_SOURCE_ROOT}/deps/w32-pthreads" "${OBS_SOURCE_ROOT}/libobs"
    )
    target_compile_definitions(${name}-pthreads PRIVATE __CLEANUP_C PTW32_BUILD)
    set_target_properties(
      ${name}-pthreads
      PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED TRUE PREFIX "" OUTPUT_NAME w32-pthreads
    )
    target_link_libraries(${name} PRIVATE ${name}-pthreads)
    set_target_properties(${name} PROPERTIES IMPORT_PREFIX "" IMPORT_SUFFIX ".lib")
  else()
    target_link_libraries(
      ${name}
      PRIVATE
        ${_common_deps}
        "-framework AppKit"
        "-framework AudioToolbox"
        "-framework AudioUnit"
        "-framework Carbon"
        "-framework Cocoa"
        "-framework CoreAudio"
        "-framework IOKit"
    )
  endif()
endfunction()

# The platform-only plugins, compile-only like libobs above: each one's own
# sources with the libobs, shared/ and deps/ headers it builds against, through
# the same precompiled headers and unity batches as the native build. This is
# where Windows- and macOS-only file-local name collisions show up.
function(_obs_cross_plugin name)
  cmake_parse_arguments(PARSE_ARGV 1 _P "OBJC_ARC;OBJC_MODULES" "DIR" "SOURCES;INCLUDES;DEFINES")
  list(TRANSFORM _P_SOURCES PREPEND "${OBS_SOURCE_ROOT}/${_P_DIR}/")
  list(TRANSFORM _P_INCLUDES PREPEND "${OBS_SOURCE_ROOT}/")
  add_library(${name} OBJECT ${_P_SOURCES})
  target_include_directories(
    ${name}
    PRIVATE
      "${OBS_SOURCE_ROOT}/${_P_DIR}"
      "${CMAKE_BINARY_DIR}/${name}"
      ${_P_INCLUDES}
      "${OBS_SOURCE_ROOT}/libobs"
      "${CMAKE_BINARY_DIR}/config"
      "${OBS_CROSS_DEPS}/include"
      $<$<STREQUAL:${_host},WINDOWS>:${OBS_SOURCE_ROOT}/deps/w32-pthreads>
  )
  target_compile_definitions(
    ${name}
    PRIVATE
      ENABLE_HEVC
      OBS_VERSION="cross-check"
      ${_P_DEFINES}
      $<$<STREQUAL:${_host},WINDOWS>:NOMINMAX;UNICODE;_UNICODE;_CRT_SECURE_NO_WARNINGS;_CRT_NONSTDC_NO_WARNINGS>
  )
  target_compile_options(
    ${name}
    PRIVATE
      "$<$<COMPILE_LANGUAGE:C,OBJC>:${_obs_clang_c_options}>"
      "$<$<COMPILE_LANGUAGE:CXX,OBJCXX>:${_obs_clang_cxx_options}>"
  )
  # The Xcode settings the plugin's CMakeLists.txt sets
  # (XCODE_ATTRIBUTE_CLANG_ENABLE_OBJC_ARC / _MODULES), as compiler flags.
  if(_P_OBJC_ARC)
    target_compile_options(${name} PRIVATE "$<$<COMPILE_LANGUAGE:OBJC,OBJCXX>:-fobjc-arc>")
  endif()
  if(_P_OBJC_MODULES)
    target_compile_options(${name} PRIVATE "$<$<COMPILE_LANGUAGE:OBJC,OBJCXX>:-fmodules>")
  endif()
  set_target_properties(
    ${name}
    PROPERTIES
      C_STANDARD 17
      C_STANDARD_REQUIRED TRUE
      CXX_STANDARD 17
      CXX_STANDARD_REQUIRED TRUE
      COMPILE_WARNING_AS_ERROR ${OBS_CROSS_WERROR}
      OBS_PCH_PROFILE plugin
  )
endfunction()

# A cross target opts into the conversion ratchet the way a native one does in
# its own CMakeLists (#2625): by naming the three groups after the shared option
# lists the declaration above already applied, so they follow `-Wno-conversion`
# and `-Wno-sign-conversion` from `_obs_clang_*_options` and win. Without it, a
# Windows- or macOS-only target is compiled by no Linux lane with the groups on
# and can never be measured (#3150). MSVC has no such flags, so the alias stays
# inert for it; clang-cl and AppleClang do honour them. `ci/conversion_ratchet.py`
# reads an opt-in from a plugin's own CMakeLists, which this is not, so the set
# of opted-in cross targets is pinned by ci/test_cross_conversion_ratchet.py.
function(_obs_cross_conversion_ratchet target)
  target_compile_options(
    ${target}
    PRIVATE
      "$<$<COMPILE_LANG_AND_ID:C,Clang>:-Wconversion;-Wsign-conversion;-Wfloat-conversion>"
      "$<$<COMPILE_LANG_AND_ID:CXX,Clang,AppleClang>:-Wconversion;-Wsign-conversion;-Wfloat-conversion>"
  )
endfunction()

# The frontend's Windows/macOS unity groups (#2408) need Qt: the target's own
# prebuilt Qt6 (CMakePresets.json `qt6`, the headers the real build uses) and
# the ui_*.h headers its forms generate. uic runs on this host, from the
# pinned Linux Qt of the same version (build-project.yaml's QT_TAG/QT_ASSET,
# kept equal by ci/test_cross_qt.py). Nothing is moc'd: moc output is compiled
# on its own, never included, and this check links nothing.
set(OBS_CROSS_HOST_QT_TAG qt6-6.11.1)
set(OBS_CROSS_HOST_QT_ASSET qt6-6.11.1-linux-x86_64-glibc2.28-release.tar.zst)
set(OBS_CROSS_HOST_QT_SHA256 f0edfc3432bd57058bbb9a2cf18aacceb55a2564b018346778c34eb0e3369dbf)
function(_obs_cross_uic out)
  set(dir "${OBS_SOURCE_ROOT}/.deps/host-${OBS_CROSS_HOST_QT_TAG}")
  if(NOT EXISTS "${dir}/.sha256")
    set(archive "${OBS_SOURCE_ROOT}/.deps/${OBS_CROSS_HOST_QT_ASSET}")
    file(
      DOWNLOAD
        "https://github.com/obs-rust/obs-studio-libs/releases/download/${OBS_CROSS_HOST_QT_TAG}/${OBS_CROSS_HOST_QT_ASSET}"
      "${archive}"
      EXPECTED_HASH SHA256=${OBS_CROSS_HOST_QT_SHA256}
      STATUS status
    )
    list(GET status 0 code)
    if(NOT code EQUAL 0)
      message(FATAL_ERROR "Downloading the host Qt for uic failed: ${status}")
    endif()
    file(REMOVE_RECURSE "${dir}")
    file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${dir}")
    file(REMOVE "${archive}")
    file(WRITE "${dir}/.sha256" "${OBS_CROSS_HOST_QT_SHA256}")
  endif()
  file(GLOB_RECURSE uic LIST_DIRECTORIES FALSE "${dir}/*/uic" "${dir}/uic")
  list(FILTER uic INCLUDE REGEX "/(bin|libexec)/uic$")
  if(NOT uic)
    message(FATAL_ERROR "No uic under ${dir}")
  endif()
  list(GET uic 0 uic)
  set(${out} "${uic}" PARENT_SCOPE)
endfunction()

# ui_<form>.h for every frontend form, into <build>/frontend-ui.
function(_obs_cross_frontend_forms)
  _obs_cross_uic(uic)
  file(GLOB forms "${OBS_SOURCE_ROOT}/frontend/forms/*.ui" "${OBS_SOURCE_ROOT}/frontend/forms/source-toolbar/*.ui")
  set(headers)
  foreach(form IN LISTS forms)
    get_filename_component(stem "${form}" NAME_WE)
    set(header "${CMAKE_BINARY_DIR}/frontend-ui/ui_${stem}.h")
    add_custom_command(OUTPUT "${header}" COMMAND "${uic}" -o "${header}" "${form}" DEPENDS "${form}" VERBATIM)
    list(APPEND headers "${header}")
  endforeach()
  add_custom_target(frontend-ui-cross DEPENDS ${headers})
endfunction()

# Include directories of the target's Qt modules the frontend uses.
function(_obs_cross_qt_includes out)
  set(qt "${OBS_CROSS_QT}")
  set(dirs)
  foreach(
    module
    IN
    ITEMS Core Gui Widgets Network Svg Xml
  )
    if(_host STREQUAL MACOS)
      list(APPEND dirs "${qt}/lib/Qt${module}.framework/Headers")
    else()
      list(APPEND dirs "${qt}/include/Qt${module}")
    endif()
  endforeach()
  # QtCore's and QtGui's private headers (the macOS frontend's qpa use).
  foreach(module IN ITEMS Core Gui)
    if(_host STREQUAL MACOS)
      set(private "${qt}/lib/Qt${module}.framework/Headers/${OBS_CROSS_QT_VERSION}")
    else()
      set(private "${qt}/include/Qt${module}/${OBS_CROSS_QT_VERSION}")
    endif()
    list(APPEND dirs "${private}" "${private}/Qt${module}")
  endforeach()
  if(NOT _host STREQUAL MACOS)
    list(APPEND dirs "${qt}/include")
  endif()
  set(${out} ${dirs} PARENT_SCOPE)
endfunction()

# One frontend group target: the Qt includes, the generated forms, and the
# include roots the frontend's own CMake gives obs-studio.
function(_obs_cross_frontend name)
  cmake_parse_arguments(PARSE_ARGV 1 _F "" "" "SOURCES;DEFINES")
  if(NOT TARGET frontend-ui-cross)
    _obs_cross_frontend_forms()
  endif()
  _obs_cross_qt_includes(qt_includes)
  _obs_cross_plugin(
    ${name}
    DIR frontend
    SOURCES ${_F_SOURCES}
    INCLUDES . frontend/api shared/qt/wrappers deps/json11
    DEFINES
      QT_ENABLE_STRICT_MODE_UP_TO=0xFF0000
      QT_CORE_LIB
      QT_GUI_LIB
      QT_WIDGETS_LIB
      QT_NETWORK_LIB
      QT_SVG_LIB
      QT_XML_LIB
      # The graphics modules' file names, as frontend/CMakeLists.txt passes them.
      DL_OPENGL="libobs-opengl"
      DL_METAL="libobs-metal"
      DL_D3D11="libobs-d3d11"
      ${_F_DEFINES}
  )
  target_include_directories(
    ${name}
    PRIVATE "${CMAKE_BINARY_DIR}/frontend-ui" "${CMAKE_BINARY_DIR}/frontend" ${qt_includes}
  )
  if(_host STREQUAL MACOS)
    target_compile_options(${name} PRIVATE "-F${OBS_CROSS_QT}/lib")
  endif()
  add_dependencies(${name} frontend-ui-cross)
endfunction()
